## Descripción del proyecto

Un lector de códigos que va más allá de mostrar texto: **entiende** lo que escanea y propone la acción adecuada. Un QR de Wi-Fi muestra la red y copia la contraseña; una URL se abre tras mostrar el dominio; un teléfono ofrece llamar.

Los códigos QR son **datos externos no confiables**: pueden llevar a sitios de phishing. Por eso mostramos el dominio real antes de abrir cualquier enlace.

```ts title="features/scanner/parse-code.ts"
export type ParsedCode =
  | { kind: 'url'; url: string; domain: string }
  | { kind: 'wifi'; ssid: string; password: string | null; security: string }
  | { kind: 'phone'; phone: string }
  | { kind: 'email'; email: string }
  | { kind: 'product'; code: string }
  | { kind: 'text'; text: string };

/** Formato estándar de QR Wi-Fi: WIFI:T:WPA;S:MiRed;P:clave;; */
function parseWifi(data: string): ParsedCode | null {
  const get = (key: string) => new RegExp(`${key}:((?:\\\\.|[^;])*)`).exec(data)?.[1]?.replace(/\\(.)/g, '$1') ?? null;
  const ssid = get('S');
  return ssid ? { kind: 'wifi', ssid, password: get('P'), security: get('T') ?? 'nopass' } : null;
}

export function parseCode(data: string, type: string): ParsedCode {
  const value = data.trim();

  if (/^https?:\/\//i.test(value)) {
    try {
      const url = new URL(value);
      return { kind: 'url', url: url.toString(), domain: url.hostname };
    } catch {
      // URL mal formada: la tratamos como texto
    }
  }
  if (/^WIFI:/i.test(value)) return parseWifi(value) ?? { kind: 'text', text: value };
  if (/^tel:/i.test(value)) return { kind: 'phone', phone: value.slice(4) };
  if (/^mailto:/i.test(value)) return { kind: 'email', email: value.slice(7).split('?')[0] };
  if (type !== 'qr' && /^\d{8,14}$/.test(value)) return { kind: 'product', code: value };
  return { kind: 'text', text: value };
}
```

```ts title="features/scanner/parse-code.test.ts"
import { parseCode } from './parse-code';

test('reconoce URLs y extrae el dominio', () => {
  expect(parseCode('https://tienda.ejemplo.com/promo?id=1', 'qr')).toEqual({
    kind: 'url',
    url: 'https://tienda.ejemplo.com/promo?id=1',
    domain: 'tienda.ejemplo.com',
  });
});

test('interpreta redes Wi-Fi con caracteres escapados', () => {
  expect(parseCode('WIFI:T:WPA;S:Oficina\\;2;P:clave123;;', 'qr')).toEqual({
    kind: 'wifi',
    ssid: 'Oficina;2',
    password: 'clave123',
    security: 'WPA',
  });
});

test('códigos de barras numéricos son productos', () => {
  expect(parseCode('7801234567890', 'ean13')).toEqual({ kind: 'product', code: '7801234567890' });
});
```

```ts title="features/scanner/scans-repository.ts"
import type { SQLiteDatabase } from 'expo-sqlite';

export type Scan = { id: number; data: string; type: string; createdAt: number };

export async function migrate(db: SQLiteDatabase) {
  await db.execAsync(`
    CREATE TABLE IF NOT EXISTS scans (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      data TEXT NOT NULL,
      type TEXT NOT NULL,
      created_at INTEGER NOT NULL
    );
  `);
}

export const scansRepository = {
  async add(db: SQLiteDatabase, data: string, type: string) {
    const r = await db.runAsync('INSERT INTO scans (data, type, created_at) VALUES (?, ?, ?)', data, type, Date.now());
    return r.lastInsertRowId;
  },
  get: (db: SQLiteDatabase, id: number) =>
    db.getFirstAsync<Scan>('SELECT id, data, type, created_at AS createdAt FROM scans WHERE id = ?', id),
  list: (db: SQLiteDatabase) =>
    db.getAllAsync<Scan>('SELECT id, data, type, created_at AS createdAt FROM scans ORDER BY created_at DESC LIMIT 200'),
  clear: (db: SQLiteDatabase) => db.runAsync('DELETE FROM scans'),
};
```

<Step title="La pantalla de escaneo">

```tsx title="src/app/index.tsx"
import { useCallback, useRef, useState } from 'react';
import { CameraView, useCameraPermissions, type BarcodeScanningResult } from 'expo-camera';
import { Link, router, useFocusEffect } from 'expo-router';
import { useSQLiteContext } from 'expo-sqlite';
import * as Haptics from 'expo-haptics';
import { Linking, View } from 'react-native';
import { Button, Text } from '@/components/ui';
import { scansRepository } from '@/features/scanner/scans-repository';

export default function ScannerScreen() {
  const db = useSQLiteContext();
  const [permission, requestPermission] = useCameraPermissions();
  const [torch, setTorch] = useState(false);
  const [active, setActive] = useState(true);
  const locked = useRef(false);

  // Reactiva el escáner al volver a esta pantalla
  useFocusEffect(
    useCallback(() => {
      locked.current = false;
      setActive(true);
      return () => setActive(false); // desmonta la cámara al salir (ahorra batería)
    }, []),
  );

  const onScanned = useCallback(
    async ({ data, type }: BarcodeScanningResult) => {
      if (locked.current) return;
      locked.current = true;
      Haptics.notificationAsync(Haptics.NotificationFeedbackType.Success);
      const id = await scansRepository.add(db, data, type);
      router.push({ pathname: '/resultado/[id]', params: { id: String(id) } });
    },
    [db],
  );

  if (!permission) return null;
  if (!permission.granted) {
    return (
      <View className="flex-1 items-center justify-center gap-4 bg-background p-6">
        <Text align="center">Para leer códigos necesitamos acceso a la cámara.</Text>
        {permission.canAskAgain ? (
          <Button onPress={requestPermission}>Permitir cámara</Button>
        ) : (
          <Button onPress={() => Linking.openSettings()}>Abrir ajustes</Button>
        )}
      </View>
    );
  }

  return (
    <View className="flex-1 bg-black">
      {active ? (
        <CameraView
          style={{ flex: 1 }}
          facing="back"
          enableTorch={torch}
          barcodeScannerSettings={{ barcodeTypes: ['qr', 'ean13', 'ean8', 'upc_a', 'code128'] }}
          onBarcodeScanned={onScanned}
        />
      ) : null}
      {/* Marco guía */}
      <View pointerEvents="none" className="absolute inset-0 items-center justify-center">
        <View className="h-64 w-64 rounded-3xl border-4 border-white/90" />
        <Text className="mt-4 text-white">Centra el código en el marco</Text>
      </View>
      <View className="absolute inset-x-6 bottom-12 flex-row gap-3">
        <Button variant="secondary" className="flex-1" onPress={() => setTorch((t) => !t)}>
          {torch ? 'Apagar linterna' : 'Linterna'}
        </Button>
        <Link href="/historial" asChild>
          <Button variant="secondary" className="flex-1">
            Historial
          </Button>
        </Link>
      </View>
    </View>
  );
}
```

<Step title="Acciones según el tipo de contenido">

```tsx title="features/scanner/components/result-actions.tsx"
import * as Clipboard from 'expo-clipboard';
import * as WebBrowser from 'expo-web-browser';
import { Alert, Linking, Share, View } from 'react-native';
import { Button, Card, Text } from '@/components/ui';
import type { ParsedCode } from '../parse-code';

export function ResultActions({ code, raw }: { code: ParsedCode; raw: string }) {
  async function copy(value: string, label = 'Copiado') {
    await Clipboard.setStringAsync(value);
    Alert.alert(label);
  }

  function openUrl(url: string, domain: string) {
    // Los QR pueden llevar a sitios maliciosos: mostramos el dominio real antes de abrir
    Alert.alert('Abrir enlace', `Vas a abrir ${domain}. ¿Confías en este sitio?`, [
      { text: 'Cancelar', style: 'cancel' },
      { text: 'Abrir', onPress: () => WebBrowser.openBrowserAsync(url) },
    ]);
  }

  return (
    <View className="gap-3">
      {code.kind === 'url' ? (
        <>
          <Card className="p-4">
            <Text variant="caption">Dominio</Text>
            <Text variant="heading">{code.domain}</Text>
          </Card>
          <Button fullWidth onPress={() => openUrl(code.url, code.domain)}>
            Abrir enlace
          </Button>
        </>
      ) : null}
      {code.kind === 'wifi' ? (
        <>
          <Card className="gap-1 p-4">
            <Text variant="heading">{code.ssid}</Text>
            <Text variant="caption">Seguridad: {code.security}</Text>
          </Card>
          {code.password ? (
            <Button fullWidth onPress={() => copy(code.password!, 'Contraseña copiada')}>
              Copiar contraseña
            </Button>
          ) : null}
        </>
      ) : null}
      {code.kind === 'phone' ? (
        <Button fullWidth onPress={() => Linking.openURL(`tel:${code.phone}`)}>
          Llamar a {code.phone}
        </Button>
      ) : null}
      {code.kind === 'email' ? (
        <Button fullWidth onPress={() => Linking.openURL(`mailto:${code.email}`)}>
          Escribir a {code.email}
        </Button>
      ) : null}
      <Button variant="secondary" fullWidth onPress={() => copy(raw)}>
        Copiar contenido
      </Button>
      <Button variant="secondary" fullWidth onPress={() => Share.share({ message: raw })}>
        Compartir
      </Button>
    </View>
  );
}
```
