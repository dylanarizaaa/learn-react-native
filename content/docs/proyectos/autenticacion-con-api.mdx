## Descripción del proyecto

Casi todas las apps de producto necesitan login. Este proyecto implementa el flujo completo con todas las piezas que un equipo profesional espera: **validación**, **tokens en SecureStore**, **refresco transparente**, **rutas protegidas** y **limpieza al cerrar sesión**.

Usamos la API pública de pruebas **DummyJSON**, que ofrece endpoints de login, perfil (`/auth/me`) y refresco con tokens reales, así el proyecto funciona sin construir un backend. Usuario de prueba de su documentación: `emilys` / `emilyspass`.

<Callout type="warning" title="Solo para aprender">
DummyJSON es una API de demostración: no envíes datos reales. En tu app, sustituye la URL base por tu backend; el resto del código no cambia.
</Callout>

```ts title="src/lib/secure-session.ts"
import * as SecureStore from 'expo-secure-store';

const ACCESS = 'auth.accessToken';
const REFRESH = 'auth.refreshToken';
const options = { keychainAccessible: SecureStore.AFTER_FIRST_UNLOCK_THIS_DEVICE_ONLY };

export type Tokens = { accessToken: string; refreshToken: string };

export async function saveTokens(t: Tokens) {
  await SecureStore.setItemAsync(ACCESS, t.accessToken, options);
  await SecureStore.setItemAsync(REFRESH, t.refreshToken, options);
}

export async function getTokens(): Promise<Tokens | null> {
  const [accessToken, refreshToken] = await Promise.all([SecureStore.getItemAsync(ACCESS, options), SecureStore.getItemAsync(REFRESH, options)]);
  return accessToken && refreshToken ? { accessToken, refreshToken } : null;
}

export async function clearTokens() {
  await Promise.all([SecureStore.deleteItemAsync(ACCESS, options), SecureStore.deleteItemAsync(REFRESH, options)]);
}
```

```ts title="src/lib/session-events.ts"
const listeners = new Set<() => void>();

export const sessionEvents = {
  emitExpired: () => listeners.forEach((l) => l()),
  onExpired: (listener: () => void) => {
    listeners.add(listener);
    return () => {
      listeners.delete(listener);
    };
  },
};
```

<Step title="Cliente HTTP con refresco automático">

Aplicamos el patrón de [JWT](/docs/datos/jwt): una única promesa de refresco compartida y un reintento por petición.

```ts title="src/lib/api-client.ts"
import axios, { type AxiosError, type InternalAxiosRequestConfig } from 'axios';
import { clearTokens, getTokens, saveTokens } from './secure-session';
import { sessionEvents } from './session-events';

const baseURL = process.env.EXPO_PUBLIC_API_URL;
export const api = axios.create({ baseURL, timeout: 15_000 });
const bare = axios.create({ baseURL, timeout: 15_000 }); // sin interceptores

let refreshing: Promise<string> | null = null;

function refreshAccessToken() {
  refreshing ??= (async () => {
    try {
      const tokens = await getTokens();
      if (!tokens) throw new Error('Sin sesión');
      const { data } = await bare.post<{ accessToken: string; refreshToken: string }>('/auth/refresh', {
        refreshToken: tokens.refreshToken,
        expiresInMins: 30,
      });
      await saveTokens(data);
      return data.accessToken;
    } catch (error) {
      await clearTokens();
      sessionEvents.emitExpired();
      throw error;
    } finally {
      refreshing = null;
    }
  })();
  return refreshing;
}

api.interceptors.request.use(async (config) => {
  const tokens = await getTokens();
  if (tokens) config.headers.Authorization = `Bearer ${tokens.accessToken}`;
  return config;
});

api.interceptors.response.use(undefined, async (error: AxiosError) => {
  const original = error.config as (InternalAxiosRequestConfig & { _retried?: boolean }) | undefined;
  if (error.response?.status !== 401 || !original || original._retried) throw error;
  original._retried = true;
  original.headers.Authorization = `Bearer ${await refreshAccessToken()}`;
  return api(original);
});
```

```ts title="src/features/auth/api/auth-api.ts"
import { api } from '@/lib/api-client';
import type { Tokens } from '@/lib/secure-session';

export type User = { id: number; username: string; email: string; firstName: string; lastName: string; image: string };

export const authApi = {
  async login(username: string, password: string): Promise<{ user: User; tokens: Tokens }> {
    const { data } = await api.post<User & Tokens>('/auth/login', { username, password, expiresInMins: 30 });
    const { accessToken, refreshToken, ...user } = data;
    return { user, tokens: { accessToken, refreshToken } };
  },
  async me(): Promise<User> {
    const { data } = await api.get<User>('/auth/me');
    return data;
  },
};
```

<Step title="Proveedor de sesión">

```tsx title="src/features/auth/session-provider.tsx"
import { useQueryClient } from '@tanstack/react-query';
import * as SplashScreen from 'expo-splash-screen';
import { createContext, use, useEffect, useMemo, useState, type ReactNode } from 'react';
import { clearTokens, getTokens, saveTokens } from '@/lib/secure-session';
import { sessionEvents } from '@/lib/session-events';
import { authApi, type User } from './api/auth-api';

type Session = { status: 'loading' } | { status: 'signed-out' } | { status: 'signed-in'; user: User };
type Ctx = { session: Session; signIn: (u: string, p: string) => Promise<void>; signOut: () => Promise<void> };

const SessionContext = createContext<Ctx | null>(null);
SplashScreen.preventAutoHideAsync();

export function SessionProvider({ children }: { children: ReactNode }) {
  const queryClient = useQueryClient();
  const [session, setSession] = useState<Session>({ status: 'loading' });

  useEffect(() => {
    (async () => {
      try {
        if (!(await getTokens())) return setSession({ status: 'signed-out' });
        setSession({ status: 'signed-in', user: await authApi.me() }); // el interceptor refresca si hace falta
      } catch {
        setSession({ status: 'signed-out' });
      } finally {
        SplashScreen.hideAsync();
      }
    })();

    // Si un refresco falla en cualquier pantalla, cerramos sesión
    return sessionEvents.onExpired(() => {
      queryClient.clear();
      setSession({ status: 'signed-out' });
    });
  }, [queryClient]);

  const value = useMemo<Ctx>(
    () => ({
      session,
      async signIn(username, password) {
        const { user, tokens } = await authApi.login(username, password);
        await saveTokens(tokens);
        setSession({ status: 'signed-in', user });
      },
      async signOut() {
        await clearTokens();
        queryClient.clear();
        setSession({ status: 'signed-out' });
      },
    }),
    [session, queryClient],
  );

  return <SessionContext value={value}>{children}</SessionContext>;
}

export function useSession() {
  const ctx = use(SessionContext);
  if (!ctx) throw new Error('useSession fuera de SessionProvider');
  return ctx;
}
```

```tsx title="src/app/_layout.tsx"
import '../../global.css';
import { QueryClient, QueryClientProvider } from '@tanstack/react-query';
import { Stack } from 'expo-router';
import { useState } from 'react';
import { SessionProvider, useSession } from '@/features/auth/session-provider';
import { ThemeProvider } from '@/theme/theme-provider';

function Navigator() {
  const { session } = useSession();
  if (session.status === 'loading') return null;
  const signedIn = session.status === 'signed-in';

  return (
    <Stack screenOptions={{ headerShown: false }}>
      <Stack.Protected guard={!signedIn}>
        <Stack.Screen name="(auth)" />
      </Stack.Protected>
      <Stack.Protected guard={signedIn} redirectTo="/login">
        <Stack.Screen name="(app)" />
      </Stack.Protected>
    </Stack>
  );
}

export default function RootLayout() {
  const [queryClient] = useState(() => new QueryClient());
  return (
    <QueryClientProvider client={queryClient}>
      <SessionProvider>
        <ThemeProvider>
          <Navigator />
        </ThemeProvider>
      </SessionProvider>
    </QueryClientProvider>
  );
}
```

```tsx title="src/app/(auth)/login.tsx"
import { zodResolver } from '@hookform/resolvers/zod';
import { isAxiosError } from 'axios';
import { useForm } from 'react-hook-form';
import { KeyboardAvoidingView, Platform, View } from 'react-native';
import { FormInput } from '@/components/form';
import { Button, Card, Text } from '@/components/ui';
import { loginSchema } from '@/features/auth/schemas';
import { useSession } from '@/features/auth/session-provider';

export default function LoginScreen() {
  const { signIn } = useSession();
  const { control, handleSubmit, setError, formState } = useForm({
    resolver: zodResolver(loginSchema),
    defaultValues: { username: '', password: '' },
  });

  const onSubmit = handleSubmit(async ({ username, password }) => {
    try {
      await signIn(username, password); // Stack.Protected navega solo al cambiar la sesión
    } catch (error) {
      const message =
        isAxiosError(error) && error.response?.status === 400 ? 'Usuario o contraseña incorrectos' : 'No pudimos conectar. Inténtalo de nuevo.';
      setError('root', { message });
    }
  });

  return (
    <KeyboardAvoidingView className="flex-1 justify-center bg-background p-6" behavior={Platform.OS === 'ios' ? 'padding' : undefined}>
      <Card>
        <Card.Header>
          <Card.Title>Bienvenido</Card.Title>
          <Card.Description>Inicia sesión para continuar</Card.Description>
        </Card.Header>
        <Card.Content className="gap-4">
          <FormInput control={control} name="username" label="Usuario" autoCapitalize="none" autoComplete="username" />
          <FormInput control={control} name="password" label="Contraseña" secureTextEntry autoComplete="current-password" />
          {formState.errors.root ? <Text tone="danger">{formState.errors.root.message}</Text> : null}
          <Button onPress={onSubmit} loading={formState.isSubmitting} fullWidth>
            Entrar
          </Button>
          <View className="items-center">
            <Text variant="caption">Prueba: emilys / emilyspass</Text>
          </View>
        </Card.Content>
      </Card>
    </KeyboardAvoidingView>
  );
}
```

## Qué has aprendido

- **Ciclo de vida de una sesión**: restaurar, usar, refrescar y cerrar.
- **Interceptores de Axios** para añadir tokens y refrescarlos con una única petición compartida.
- **`Stack.Protected`** para que la navegación dependa del estado de la sesión, también con deep links.
- **Errores del servidor en formularios** con `setError('root')`.
- **Higiene de seguridad**: SecureStore, limpieza de caché y tokens al salir.
