## Descripción del proyecto

Pequeños comercios, almacenes y talleres necesitan saber **qué tienen y cuánto**. Esta app permite registrar productos, sumar y restar stock con un historial de movimientos, y encontrar un producto al instante **escaneando su código de barras**.

El reto técnico principal es la **consistencia**: una salida de stock debe registrar el movimiento **y** actualizar el stock a la vez, y nunca dejar el stock en negativo. Lo resolvemos con **transacciones de SQLite**. Para la UI usamos el [kit de componentes](/docs/sistema-de-componentes/vision-general) y TanStack Query como capa de caché sobre la base de datos local.

```ts title="features/inventory/db/migrations.ts"
import type { SQLiteDatabase } from 'expo-sqlite';

export async function migrate(db: SQLiteDatabase) {
  const row = await db.getFirstAsync<{ user_version: number }>('PRAGMA user_version');
  if ((row?.user_version ?? 0) >= 1) return;

  await db.execAsync(`
    PRAGMA journal_mode = WAL;
    PRAGMA foreign_keys = ON;
    CREATE TABLE products (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      sku TEXT NOT NULL UNIQUE,
      name TEXT NOT NULL,
      category TEXT NOT NULL,
      price REAL NOT NULL DEFAULT 0,
      stock INTEGER NOT NULL DEFAULT 0 CHECK (stock >= 0),
      min_stock INTEGER NOT NULL DEFAULT 0
    );
    CREATE TABLE movements (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      product_id INTEGER NOT NULL REFERENCES products (id) ON DELETE CASCADE,
      type TEXT NOT NULL CHECK (type IN ('in', 'out')),
      quantity INTEGER NOT NULL CHECK (quantity > 0),
      note TEXT,
      created_at INTEGER NOT NULL
    );
    CREATE INDEX idx_movements_product ON movements (product_id, created_at DESC);
    PRAGMA user_version = 1;
  `);
}
```

```ts title="features/inventory/db/inventory-repository.ts"
import type { SQLiteDatabase } from 'expo-sqlite';

export type Product = { id: number; sku: string; name: string; category: string; price: number; stock: number; minStock: number };
export type Movement = { id: number; type: 'in' | 'out'; quantity: number; note: string | null; createdAt: number };
type ProductRow = Omit<Product, 'minStock'> & { min_stock: number };

const toProduct = ({ min_stock, ...rest }: ProductRow): Product => ({ ...rest, minStock: min_stock });

export class InsufficientStockError extends Error {
  constructor(public available: number) {
    super(`Solo hay ${available} unidades disponibles`);
  }
}

export const inventoryRepository = {
  async list(db: SQLiteDatabase, { search = '', lowOnly = false }) {
    const rows = await db.getAllAsync<ProductRow>(
      `SELECT * FROM products
       WHERE (name LIKE $q OR sku LIKE $q) AND ($low = 0 OR stock <= min_stock)
       ORDER BY (stock <= min_stock) DESC, name`,
      { $q: `%${search}%`, $low: lowOnly ? 1 : 0 },
    );
    return rows.map(toProduct);
  },
  async get(db: SQLiteDatabase, id: number) {
    const row = await db.getFirstAsync<ProductRow>('SELECT * FROM products WHERE id = ?', id);
    return row ? toProduct(row) : null;
  },
  async findBySku(db: SQLiteDatabase, sku: string) {
    const row = await db.getFirstAsync<ProductRow>('SELECT * FROM products WHERE sku = ?', sku);
    return row ? toProduct(row) : null;
  },
  async save(db: SQLiteDatabase, p: Omit<Product, 'id' | 'stock'> & { id?: number }) {
    if (p.id) {
      await db.runAsync('UPDATE products SET sku=?, name=?, category=?, price=?, min_stock=? WHERE id=?', p.sku, p.name, p.category, p.price, p.minStock, p.id);
      return p.id;
    }
    const r = await db.runAsync('INSERT INTO products (sku, name, category, price, min_stock) VALUES (?,?,?,?,?)', p.sku, p.name, p.category, p.price, p.minStock);
    return r.lastInsertRowId;
  },
  movements(db: SQLiteDatabase, productId: number) {
    return db.getAllAsync<Movement>(
      'SELECT id, type, quantity, note, created_at AS createdAt FROM movements WHERE product_id = ? ORDER BY created_at DESC LIMIT 50',
      productId,
    );
  },
  /** Registra el movimiento y actualiza el stock de forma atómica */
  async registerMovement(db: SQLiteDatabase, input: { productId: number; type: 'in' | 'out'; quantity: number; note?: string }) {
    await db.withExclusiveTransactionAsync(async (tx) => {
      const product = await tx.getFirstAsync<{ stock: number }>('SELECT stock FROM products WHERE id = ?', input.productId);
      if (!product) throw new Error('Producto no encontrado');
      const delta = input.type === 'in' ? input.quantity : -input.quantity;
      if (product.stock + delta < 0) throw new InsufficientStockError(product.stock);

      await tx.runAsync('UPDATE products SET stock = stock + ? WHERE id = ?', delta, input.productId);
      await tx.runAsync(
        'INSERT INTO movements (product_id, type, quantity, note, created_at) VALUES (?, ?, ?, ?, ?)',
        input.productId, input.type, input.quantity, input.note ?? null, Date.now(),
      );
    });
  },
};
```

<Step title="Hooks con TanStack Query sobre SQLite">

TanStack Query no es solo para APIs: también cachea consultas locales y gestiona la invalidación tras cada mutación.

```ts title="features/inventory/hooks/use-inventory.ts"
import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query';
import { useSQLiteContext } from 'expo-sqlite';
import { inventoryRepository, type Product } from '../db/inventory-repository';

const keys = {
  all: ['inventory'] as const,
  list: (f: { search: string; lowOnly: boolean }) => [...keys.all, 'list', f] as const,
  product: (id: number) => [...keys.all, 'product', id] as const,
  movements: (id: number) => [...keys.all, 'movements', id] as const,
};

export function useProducts(filters: { search: string; lowOnly: boolean }) {
  const db = useSQLiteContext();
  return useQuery({ queryKey: keys.list(filters), queryFn: () => inventoryRepository.list(db, filters) });
}

export function useProduct(id: number) {
  const db = useSQLiteContext();
  return useQuery({ queryKey: keys.product(id), queryFn: () => inventoryRepository.get(db, id) });
}

export function useMovements(id: number) {
  const db = useSQLiteContext();
  return useQuery({ queryKey: keys.movements(id), queryFn: () => inventoryRepository.movements(db, id) });
}

export function useRegisterMovement() {
  const db = useSQLiteContext();
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: (input: Parameters<typeof inventoryRepository.registerMovement>[1]) => inventoryRepository.registerMovement(db, input),
    onSuccess: () => queryClient.invalidateQueries({ queryKey: keys.all }),
  });
}

export function useSaveProduct() {
  const db = useSQLiteContext();
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: (p: Omit<Product, 'id' | 'stock'> & { id?: number }) => inventoryRepository.save(db, p),
    onSuccess: () => queryClient.invalidateQueries({ queryKey: keys.all }),
  });
}
```

```tsx title="src/app/producto/[id].tsx"
import { useState } from 'react';
import { Stack, useLocalSearchParams } from 'expo-router';
import { Alert, FlatList, View } from 'react-native';
import { LoadingState } from '@/components/feedback';
import { Button, Card, Input, SegmentedControl, Text } from '@/components/ui';
import { InsufficientStockError } from '@/features/inventory/db/inventory-repository';
import { useMovements, useProduct, useRegisterMovement } from '@/features/inventory/hooks/use-inventory';

export default function ProductDetail() {
  const id = Number(useLocalSearchParams<{ id: string }>().id);
  const { data: product, isPending } = useProduct(id);
  const { data: movements = [] } = useMovements(id);
  const register = useRegisterMovement();
  const [type, setType] = useState<'in' | 'out'>('in');
  const [quantity, setQuantity] = useState('1');
  const [note, setNote] = useState('');

  if (isPending || !product) return <LoadingState />;

  function submit() {
    const qty = Number.parseInt(quantity, 10);
    if (!Number.isInteger(qty) || qty <= 0) return Alert.alert('Cantidad no válida');
    register.mutate(
      { productId: id, type, quantity: qty, note: note.trim() || undefined },
      {
        onSuccess: () => {
          setQuantity('1');
          setNote('');
        },
        onError: (error) =>
          Alert.alert('No se pudo registrar', error instanceof InsufficientStockError ? error.message : 'Inténtalo de nuevo.'),
      },
    );
  }

  return (
    <FlatList
      className="flex-1 bg-background"
      contentContainerClassName="gap-3 p-4"
      data={movements}
      keyExtractor={(m) => String(m.id)}
      ListHeaderComponent={
        <View className="gap-4 pb-2">
          <Stack.Screen options={{ title: product.name }} />
          <View className="flex-row gap-3">
            <Card className="flex-1 p-4">
              <Text variant="caption">Stock</Text>
              <Text variant="title" tone={product.stock <= product.minStock ? 'danger' : 'default'}>
                {product.stock}
              </Text>
            </Card>
            <Card className="flex-1 p-4">
              <Text variant="caption">Mínimo</Text>
              <Text variant="title">{product.minStock}</Text>
            </Card>
          </View>
          <SegmentedControl
            accessibilityLabel="Tipo de movimiento"
            options={[
              { value: 'in', label: 'Entrada' },
              { value: 'out', label: 'Salida' },
            ]}
            value={type}
            onChange={setType}
          />
          <Input label="Cantidad" value={quantity} onChangeText={setQuantity} inputMode="numeric" />
          <Input label="Nota (opcional)" value={note} onChangeText={setNote} placeholder="Proveedor, venta, merma…" />
          <Button onPress={submit} loading={register.isPending} fullWidth>
            Registrar {type === 'in' ? 'entrada' : 'salida'}
          </Button>
          <Text variant="heading" className="mt-2">
            Movimientos
          </Text>
        </View>
      }
      renderItem={({ item }) => (
        <View className="flex-row justify-between border-b border-border py-2">
          <View>
            <Text>{item.type === 'in' ? 'Entrada' : 'Salida'}</Text>
            {item.note ? <Text variant="caption">{item.note}</Text> : null}
          </View>
          <Text tone={item.type === 'in' ? 'success' : 'danger'} className="font-semibold">
            {item.type === 'in' ? '+' : '−'}
            {item.quantity}
          </Text>
        </View>
      )}
    />
  );
}
```

```tsx title="src/app/escanear.tsx"
import { useRef } from 'react';
import { CameraView, useCameraPermissions } from 'expo-camera';
import { router } from 'expo-router';
import { useSQLiteContext } from 'expo-sqlite';
import * as Haptics from 'expo-haptics';
import { View } from 'react-native';
import { Button, Text } from '@/components/ui';
import { inventoryRepository } from '@/features/inventory/db/inventory-repository';

export default function ScanScreen() {
  const db = useSQLiteContext();
  const [permission, requestPermission] = useCameraPermissions();
  const handled = useRef(false);

  if (!permission) return null;
  if (!permission.granted) {
    return (
      <View className="flex-1 items-center justify-center gap-4 bg-background p-6">
        <Text align="center">Necesitamos la cámara para leer códigos de barras.</Text>
        <Button onPress={requestPermission}>Permitir cámara</Button>
      </View>
    );
  }

  async function onScanned({ data }: { data: string }) {
    if (handled.current) return; // el evento llega muchas veces por segundo
    handled.current = true;
    Haptics.notificationAsync(Haptics.NotificationFeedbackType.Success);
    const product = await inventoryRepository.findBySku(db, data);
    if (product) router.replace({ pathname: '/producto/[id]', params: { id: String(product.id) } });
    else router.replace({ pathname: '/producto/nuevo', params: { sku: data } });
  }

  return (
    <View className="flex-1 bg-black">
      <CameraView
        style={{ flex: 1 }}
        facing="back"
        barcodeScannerSettings={{ barcodeTypes: ['ean13', 'ean8', 'upc_a', 'code128', 'qr'] }}
        onBarcodeScanned={onScanned}
      />
      <View className="absolute inset-x-0 bottom-12 items-center">
        <Text className="text-white">Apunta al código de barras del producto</Text>
      </View>
    </View>
  );
}
```
