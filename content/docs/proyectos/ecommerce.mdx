## Descripción del proyecto

Este es el proyecto integrador: una tienda con todo lo que espera un usuario de una app de comercio moderna. No introduce conceptos nuevos, sino que **combina** los anteriores con una arquitectura que podría mantener un equipo durante años.

El punto crítico es el **pago**: la app **nunca** calcula el precio final ni maneja claves secretas. Envía el contenido del carrito a tu backend, que recalcula el total, crea un *PaymentIntent* en Stripe y devuelve solo un secreto de cliente de un solo uso.

```ts title="features/catalog/hooks.ts"
import { keepPreviousData, useInfiniteQuery, useQuery } from '@tanstack/react-query';
import { catalogApi, type CatalogFilters } from './api';

export const catalogKeys = {
  all: ['catalog'] as const,
  list: (f: CatalogFilters) => [...catalogKeys.all, 'list', f] as const,
  product: (id: string) => [...catalogKeys.all, 'product', id] as const,
};

export function useCatalog(filters: CatalogFilters) {
  return useInfiniteQuery({
    queryKey: catalogKeys.list(filters),
    queryFn: ({ pageParam, signal }) => catalogApi.list({ ...filters, cursor: pageParam }, signal),
    initialPageParam: undefined as string | undefined,
    getNextPageParam: (last) => last.nextCursor ?? undefined,
    placeholderData: keepPreviousData,
    select: (data) => data.pages.flatMap((p) => p.items),
  });
}

export function useProduct(id: string) {
  return useQuery({ queryKey: catalogKeys.product(id), queryFn: ({ signal }) => catalogApi.get(id, signal) });
}
```

import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query';
import { api } from '@/lib/api-client';

const key = ['favorites'] as const;

export function useFavoriteIds() {
  return useQuery({
    queryKey: key,
    queryFn: async () => new Set((await api.get<string[]>('/me/favorites')).data),
    staleTime: 5 * 60_000,
  });
}

export function useToggleFavorite() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: ({ productId, favorite }: { productId: string; favorite: boolean }) =>
      favorite ? api.put(`/me/favorites/${productId}`) : api.delete(`/me/favorites/${productId}`),
    onMutate: async ({ productId, favorite }) => {
      await queryClient.cancelQueries({ queryKey: key });
      const previous = queryClient.getQueryData<Set<string>>(key);
      const next = new Set(previous);
      if (favorite) next.add(productId);
      else next.delete(productId);
      queryClient.setQueryData(key, next); // el corazón cambia al instante
      return { previous };
    },
    onError: (_e, _v, onMutateResult) => queryClient.setQueryData(key, onMutateResult?.previous),
    onSettled: () => queryClient.invalidateQueries({ queryKey: key }),
  });
}

```ts title="features/cart/store.ts"
import AsyncStorage from '@react-native-async-storage/async-storage';
import { create } from 'zustand';
import { createJSONStorage, persist } from 'zustand/middleware';

export type CartLine = { productId: string; name: string; price: number; image: string; size: string | null; quantity: number };
const lineKey = (l: Pick<CartLine, 'productId' | 'size'>) => `${l.productId}:${l.size ?? ''}`;

type CartState = {
  lines: CartLine[];
  add: (item: Omit<CartLine, 'quantity'>) => void;
  setQuantity: (key: string, quantity: number) => void;
  clear: () => void;
};

export const useCartStore = create<CartState>()(
  persist(
    (set) => ({
      lines: [],
      add: (item) =>
        set((s) => {
          const k = lineKey(item);
          const existing = s.lines.find((l) => lineKey(l) === k);
          return existing
            ? { lines: s.lines.map((l) => (lineKey(l) === k ? { ...l, quantity: Math.min(l.quantity + 1, 10) } : l)) }
            : { lines: [...s.lines, { ...item, quantity: 1 }] };
        }),
      setQuantity: (k, quantity) =>
        set((s) => ({
          lines: quantity <= 0 ? s.lines.filter((l) => lineKey(l) !== k) : s.lines.map((l) => (lineKey(l) === k ? { ...l, quantity } : l)),
        })),
      clear: () => set({ lines: [] }),
    }),
    { name: 'shop-cart-v1', storage: createJSONStorage(() => AsyncStorage) },
  ),
);

export { lineKey };
export const selectSubtotal = (s: CartState) => s.lines.reduce((sum, l) => sum + l.price * l.quantity, 0);
export const selectItemCount = (s: CartState) => s.lines.reduce((n, l) => n + l.quantity, 0);
```

```ts title="features/checkout/schema.ts"
import { z } from 'zod';

export const shippingSchema = z.object({
  fullName: z.string().trim().min(3, 'Escribe tu nombre completo'),
  street: z.string().trim().min(5, 'Escribe la dirección'),
  city: z.string().trim().min(2, 'Escribe la comuna o ciudad'),
  phone: z.string().regex(/^\+?\d{8,15}$/, 'Teléfono no válido'),
});

export type ShippingValues = z.infer<typeof shippingSchema>;
```

```ts title="features/checkout/use-checkout.ts"
import { useStripe } from '@stripe/stripe-react-native';
import { useQueryClient } from '@tanstack/react-query';
import { useState } from 'react';
import { api } from '@/lib/api-client';
import { useCartStore } from '@/features/cart/store';
import type { ShippingValues } from './schema';

type IntentResponse = { orderId: string; clientSecret: string; total: number };

export function useCheckout() {
  const { initPaymentSheet, presentPaymentSheet } = useStripe();
  const queryClient = useQueryClient();
  const [status, setStatus] = useState<'idle' | 'processing' | 'success' | 'error'>('idle');

  async function pay(shipping: ShippingValues): Promise<{ ok: true; orderId: string } | { ok: false; message: string }> {
    setStatus('processing');
    try {
      const lines = useCartStore.getState().lines;
      // 1. El backend valida stock y precios, crea el pedido y el PaymentIntent
      const { data } = await api.post<IntentResponse>('/payments/intent', {
        items: lines.map((l) => ({ productId: l.productId, size: l.size, quantity: l.quantity })),
        shipping,
      });

      // 2. Preparamos la hoja de pago nativa de Stripe
      const init = await initPaymentSheet({
        merchantDisplayName: 'Mi Tienda',
        paymentIntentClientSecret: data.clientSecret,
        returnURL: 'tienda://checkout',
        defaultBillingDetails: { name: shipping.fullName, phone: shipping.phone },
      });
      if (init.error) throw new Error(init.error.message);

      // 3. El usuario introduce la tarjeta (o Apple Pay / Google Pay) en la UI de Stripe
      const result = await presentPaymentSheet();
      if (result.error) {
        setStatus('idle');
        return { ok: false, message: result.error.code === 'Canceled' ? 'Pago cancelado' : result.error.message };
      }

      // 4. Éxito: vaciamos el carrito; el backend confirmará el pedido vía webhook
      useCartStore.getState().clear();
      queryClient.invalidateQueries({ queryKey: ['orders'] });
      setStatus('success');
      return { ok: true, orderId: data.orderId };
    } catch (error) {
      setStatus('error');
      return { ok: false, message: error instanceof Error ? error.message : 'No se pudo completar el pago' };
    }
  }

  return { pay, status };
}
```

<Callout type="info" title="El backend del pago (resumen)">
El endpoint `POST /payments/intent` debe: autenticar al usuario, recalcular el total con los precios de tu base de datos, reservar stock, crear el pedido en estado `pending` y crear un *PaymentIntent* con la clave secreta de Stripe. Un *webhook* `payment_intent.succeeded` marca el pedido como pagado. La app solo muestra el estado.
</Callout>

## Qué has aprendido

- **Arquitectura por funcionalidades** en una app grande, con rutas delgadas.
- **Paginación infinita** con `select` para aplanar páginas y `keepPreviousData` al cambiar filtros.
- **Deep links** gratuitos gracias a Expo Router y enlaces compartibles con `createURL`.
- **Actualizaciones optimistas** con reversión en caso de error.
- **Pagos seguros**: el cliente nunca fija precios ni usa claves secretas; el backend y los webhooks deciden.
