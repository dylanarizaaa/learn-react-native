## Descripción del proyecto

Un **punto de venta (POS)** para ferias, cafeterías o comercios pequeños: el vendedor toca productos, ajusta cantidades, cobra y la venta queda registrada. Al final del día consulta el resumen y exporta las ventas.

La lección central es **separar los dos tipos de estado**:

- **Estado del servidor** (catálogo, ventas registradas) → **TanStack Query**.
- **Estado del cliente** (el carrito en curso) → **Zustand**, persistido para no perderlo si la app se cierra.

Usaremos **json-server** como API de desarrollo para que el proyecto funcione de extremo a extremo sin escribir un backend.

```ts title="features/sales/store/cart-store.ts"
import AsyncStorage from '@react-native-async-storage/async-storage';
import { create } from 'zustand';
import { createJSONStorage, persist } from 'zustand/middleware';

export type Product = { id: string; name: string; category: string; price: number };
export type CartLine = { product: Product; quantity: number };

type CartState = {
  lines: CartLine[];
  add: (product: Product) => void;
  setQuantity: (productId: string, quantity: number) => void;
  clear: () => void;
};

export const useCartStore = create<CartState>()(
  persist(
    (set) => ({
      lines: [],
      add: (product) =>
        set((s) => {
          const exists = s.lines.some((l) => l.product.id === product.id);
          return {
            lines: exists
              ? s.lines.map((l) => (l.product.id === product.id ? { ...l, quantity: l.quantity + 1 } : l))
              : [...s.lines, { product, quantity: 1 }],
          };
        }),
      setQuantity: (productId, quantity) =>
        set((s) => ({
          lines: quantity <= 0 ? s.lines.filter((l) => l.product.id !== productId) : s.lines.map((l) => (l.product.id === productId ? { ...l, quantity } : l)),
        })),
      clear: () => set({ lines: [] }),
    }),
    { name: 'pos-cart', storage: createJSONStorage(() => AsyncStorage) },
  ),
);

export const selectTotal = (s: CartState) => s.lines.reduce((sum, l) => sum + l.product.price * l.quantity, 0);
export const selectCount = (s: CartState) => s.lines.reduce((n, l) => n + l.quantity, 0);
```

```ts title="features/sales/api/sales-api.ts"
import { httpClient } from '@/lib/http-client';
import type { CartLine, Product } from '../store/cart-store';

export type PaymentMethod = 'cash' | 'card';
export type Sale = {
  id: string; // generado en el cliente: idempotencia
  date: string; // YYYY-MM-DD
  createdAt: string;
  method: PaymentMethod;
  total: number;
  lines: { productId: string; name: string; price: number; quantity: number }[];
};

export const salesApi = {
  products: (signal?: AbortSignal) => httpClient.get<Product[]>('/products', signal),
  salesOf: (date: string, signal?: AbortSignal) => httpClient.get<Sale[]>(`/sales?date=${date}`, signal),
  create: (sale: Sale) => httpClient.post<Sale>('/sales', sale),
};

export function buildSale(lines: CartLine[], method: PaymentMethod): Sale {
  const now = new Date();
  return {
    id: `${now.getTime()}-${Math.random().toString(36).slice(2, 8)}`,
    date: now.toISOString().slice(0, 10),
    createdAt: now.toISOString(),
    method,
    total: lines.reduce((s, l) => s + l.product.price * l.quantity, 0),
    lines: lines.map((l) => ({ productId: l.product.id, name: l.product.name, price: l.product.price, quantity: l.quantity })),
  };
}
```

```ts title="features/sales/hooks/use-sales.ts"
import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query';
import { salesApi, type Sale } from '../api/sales-api';

const today = () => new Date().toISOString().slice(0, 10);

export function useProducts() {
  return useQuery({ queryKey: ['products'], queryFn: ({ signal }) => salesApi.products(signal), staleTime: 5 * 60_000 });
}

export function useTodaySales() {
  const date = today();
  return useQuery({ queryKey: ['sales', date], queryFn: ({ signal }) => salesApi.salesOf(date, signal) });
}

export function useCreateSale() {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: (sale: Sale) => salesApi.create(sale),
    // El id se genera en el cliente: si se reintenta, el servidor puede detectar el duplicado
    retry: 2,
    onSuccess: (sale) => queryClient.invalidateQueries({ queryKey: ['sales', sale.date] }),
  });
}
```

```tsx title="src/app/(tabs)/index.tsx"
export default function SellScreen() {
  const { data: products, isPending, isError, refetch } = useProducts();
  const [category, setCategory] = useState('Todo');
  const add = useCartStore((s) => s.add);
```

```tsx title="src/app/cobrar.tsx"
import { useState } from 'react';
import { router } from 'expo-router';
import { Alert, ScrollView, View } from 'react-native';
import { Button, Input, SegmentedControl, Text } from '@/components/ui';
import { buildSale, type PaymentMethod } from '@/features/sales/api/sales-api';
import { useCreateSale } from '@/features/sales/hooks/use-sales';
import { selectTotal, useCartStore } from '@/features/sales/store/cart-store';

const money = new Intl.NumberFormat('es-CL', { style: 'currency', currency: 'CLP' });

export default function CheckoutScreen() {
  const lines = useCartStore((s) => s.lines);
  const setQuantity = useCartStore((s) => s.setQuantity);
  const clear = useCartStore((s) => s.clear);
  const total = useCartStore(selectTotal);
  const [method, setMethod] = useState<PaymentMethod>('cash');
  const [received, setReceived] = useState('');
  const createSale = useCreateSale();

  const receivedAmount = Number(received.replace(/\D/g, '')) || 0;
  const change = receivedAmount - total;
  const canConfirm = lines.length > 0 && (method === 'card' || change >= 0);

  function confirm() {
    createSale.mutate(buildSale(lines, method), {
      onSuccess: () => {
        clear();
        router.back();
      },
      onError: () => Alert.alert('No se registró la venta', 'Comprueba la conexión. El carrito se conserva.'),
    });
  }

  return (
    <ScrollView className="flex-1 bg-background" contentContainerClassName="gap-4 p-5">
      {lines.map((l) => (
        <View key={l.product.id} className="flex-row items-center gap-3">
          <Text className="flex-1">{l.product.name}</Text>
          <Button variant="outline" size="sm" onPress={() => setQuantity(l.product.id, l.quantity - 1)} accessibilityLabel={`Quitar un ${l.product.name}`}>
            −
          </Button>
          <Text className="w-6 text-center font-semibold">{l.quantity}</Text>
          <Button variant="outline" size="sm" onPress={() => setQuantity(l.product.id, l.quantity + 1)} accessibilityLabel={`Añadir un ${l.product.name}`}>
            +
          </Button>
        </View>
      ))}
      <View className="flex-row justify-between border-t border-border pt-4">
        <Text variant="heading">Total</Text>
        <Text variant="heading">{money.format(total)}</Text>
      </View>
      <SegmentedControl
        accessibilityLabel="Método de pago"
        options={[
          { value: 'cash', label: 'Efectivo' },
          { value: 'card', label: 'Tarjeta' },
        ]}
        value={method}
        onChange={setMethod}
      />
      {method === 'cash' ? (
        <>
          <Input label="Recibido" value={received} onChangeText={setReceived} inputMode="numeric" placeholder="0" />
          <Text tone={change < 0 ? 'danger' : 'success'} className="text-right text-lg font-semibold">
            {change < 0 ? `Faltan ${money.format(-change)}` : `Cambio: ${money.format(change)}`}
          </Text>
        </>
      ) : null}
      <Button size="lg" fullWidth disabled={!canConfirm} loading={createSale.isPending} onPress={confirm}>
        Confirmar venta
      </Button>
    </ScrollView>
  );
}
```

```ts title="features/sales/domain/summary.ts"
import type { Sale } from '../api/sales-api';

export function summarize(sales: Sale[]) {
  const total = sales.reduce((s, sale) => s + sale.total, 0);
  const card = sales.filter((s) => s.method === 'card').reduce((s, sale) => s + sale.total, 0);
  const units = new Map<string, number>();
  for (const sale of sales) for (const line of sale.lines) units.set(line.name, (units.get(line.name) ?? 0) + line.quantity);

  return {
    total,
    tickets: sales.length,
    average: sales.length ? Math.round(total / sales.length) : 0,
    cardShare: total ? Math.round((card / total) * 100) : 0,
    topProducts: [...units.entries()].sort((a, b) => b[1] - a[1]).slice(0, 5),
  };
}
```

```ts title="features/sales/domain/csv.ts"
import { File, Paths } from 'expo-file-system';
import * as Sharing from 'expo-sharing';
import type { Sale } from '../api/sales-api';

const cell = (v: string | number) => (/[",\n]/.test(String(v)) ? `"${String(v).replace(/"/g, '""')}"` : String(v));

export async function exportSalesCsv(date: string, sales: Sale[]) {
  const header = 'hora,producto,cantidad,precio,metodo';
  const rows = sales.flatMap((s) =>
    s.lines.map((l) => [s.createdAt.slice(11, 16), l.name, l.quantity, l.price, s.method === 'card' ? 'tarjeta' : 'efectivo'].map(cell).join(',')),
  );
  const file = new File(Paths.cache, `ventas-${date}.csv`);
  if (file.exists) file.delete();
  file.create();
  await file.write('﻿' + [header, ...rows].join('\n'));
  await Sharing.shareAsync(file.uri, { mimeType: 'text/csv', dialogTitle: `Ventas del ${date}` });
}
```
