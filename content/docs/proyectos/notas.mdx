## Descripción del proyecto

Una app de notas es el paso natural después de la lista de tareas: añade **varias pantallas**, **parámetros en la URL** y una **base de datos real**. Las notas se guardan en SQLite, así que la app funciona sin conexión y puede manejar miles de notas con búsqueda instantánea.

```ts title="src/db/migrations.ts"
import type { SQLiteDatabase } from 'expo-sqlite';

const VERSION = 1;

export async function migrate(db: SQLiteDatabase) {
  const row = await db.getFirstAsync<{ user_version: number }>('PRAGMA user_version');
  const current = row?.user_version ?? 0;
  if (current >= VERSION) return;

  if (current < 1) {
    await db.execAsync(`
      PRAGMA journal_mode = WAL;
      CREATE TABLE notes (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        title TEXT NOT NULL DEFAULT '',
        body TEXT NOT NULL DEFAULT '',
        pinned INTEGER NOT NULL DEFAULT 0,
        updated_at INTEGER NOT NULL
      );
      CREATE INDEX idx_notes_updated ON notes (pinned DESC, updated_at DESC);
    `);
  }
  await db.execAsync(`PRAGMA user_version = ${VERSION}`);
}
```

```ts title="src/db/notes-repository.ts"
import type { SQLiteDatabase } from 'expo-sqlite';

export type Note = { id: number; title: string; body: string; pinned: boolean; updatedAt: number };
type Row = { id: number; title: string; body: string; pinned: number; updated_at: number };

const toNote = (r: Row): Note => ({ id: r.id, title: r.title, body: r.body, pinned: r.pinned === 1, updatedAt: r.updated_at });

export const notesRepository = {
  async list(db: SQLiteDatabase, search: string) {
    const rows = await db.getAllAsync<Row>(
      `SELECT * FROM notes
       WHERE title LIKE $q OR body LIKE $q
       ORDER BY pinned DESC, updated_at DESC`,
      { $q: `%${search.trim()}%` },
    );
    return rows.map(toNote);
  },
  async get(db: SQLiteDatabase, id: number) {
    const row = await db.getFirstAsync<Row>('SELECT * FROM notes WHERE id = ?', id);
    return row ? toNote(row) : null;
  },
  async create(db: SQLiteDatabase, title: string, body: string) {
    const result = await db.runAsync('INSERT INTO notes (title, body, updated_at) VALUES (?, ?, ?)', title, body, Date.now());
    return result.lastInsertRowId;
  },
  update(db: SQLiteDatabase, id: number, fields: { title: string; body: string }) {
    return db.runAsync('UPDATE notes SET title = ?, body = ?, updated_at = ? WHERE id = ?', fields.title, fields.body, Date.now(), id);
  },
  togglePin(db: SQLiteDatabase, id: number) {
    return db.runAsync('UPDATE notes SET pinned = 1 - pinned WHERE id = ?', id);
  },
  remove(db: SQLiteDatabase, id: number) {
    return db.runAsync('DELETE FROM notes WHERE id = ?', id);
  },
};
```

```ts title="src/hooks/use-notes.ts"
import { useCallback, useState } from 'react';
import { useFocusEffect } from 'expo-router';
import { useSQLiteContext } from 'expo-sqlite';
import { notesRepository, type Note } from '@/db/notes-repository';

export function useNotes(search: string) {
  const db = useSQLiteContext();
  const [notes, setNotes] = useState<Note[]>([]);

  const reload = useCallback(() => {
    notesRepository.list(db, search).then(setNotes);
  }, [db, search]);

  // Se recarga al volver a la lista (tras crear o editar) y al cambiar la búsqueda
  useFocusEffect(reload);

  return { notes, reload };
}
```

```ts title="src/hooks/use-debounced-callback.ts"
import { useEffect, useRef } from 'react';

/** Ejecuta fn cuando pasan `delay` ms sin nuevas llamadas */
export function useDebouncedCallback<A extends unknown[]>(fn: (...args: A) => void, delay = 600) {
  const timer = useRef<ReturnType<typeof setTimeout> | null>(null);
  const fnRef = useRef(fn);
  fnRef.current = fn;

  useEffect(() => () => {
    if (timer.current) clearTimeout(timer.current);
  }, []);

  return (...args: A) => {
    if (timer.current) clearTimeout(timer.current);
    timer.current = setTimeout(() => fnRef.current(...args), delay);
  };
}
```

import { Link, Stack, router } from 'expo-router';

import { notesRepository } from '@/db/notes-repository';

const formatter = new Intl.DateTimeFormat('es', { day: 'numeric', month: 'short' });

export default function NotesList() {
  const db = useSQLiteContext();
  const [search, setSearch] = useState('');
  const { notes, reload } = useNotes(search);

  return (
    <>
      <Stack.Screen
        options={{
          headerSearchBarOptions: { placeholder: 'Buscar notas…', onChangeText: (e) => setSearch(e.nativeEvent.text) },
          headerRight: () => (

import { useState } from 'react';
import { Stack, router } from 'expo-router';
import { useSQLiteContext } from 'expo-sqlite';
import { Pressable, StyleSheet, Text, TextInput, View } from 'react-native';
import { notesRepository } from '@/db/notes-repository';

export default function NewNote() {
  const db = useSQLiteContext();
  const [title, setTitle] = useState('');
  const [body, setBody] = useState('');

  async function save() {
    const id = await notesRepository.create(db, title.trim(), body.trim());
    router.dismiss(); // cierra el modal…
    router.push({ pathname: '/nota/[id]', params: { id: String(id) } }); // …y abre la nota
  }

  return (
    <View style={styles.container}>
      <Stack.Screen
        options={{
          headerRight: () => (
            <Pressable onPress={save} disabled={!title.trim() && !body.trim()} hitSlop={8}>
              <Text style={styles.save}>Guardar</Text>
            </Pressable>
          ),
        }}
      />
      <TextInput value={title} onChangeText={setTitle} placeholder="Título" style={styles.title} autoFocus />
      <TextInput value={body} onChangeText={setBody} placeholder="Escribe tu nota…" multiline style={styles.body} textAlignVertical="top" />
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, padding: 20, gap: 12, backgroundColor: '#fff' },
  save: { color: '#0284c7', fontSize: 17, fontWeight: '600' },
  title: { fontSize: 22, fontWeight: '700' },
  body: { flex: 1, fontSize: 17, lineHeight: 24 },
});

```tsx title="src/app/nota/[id].tsx"
import { useEffect, useState } from 'react';
import { Stack, router, useLocalSearchParams } from 'expo-router';
import { useSQLiteContext } from 'expo-sqlite';
import { Alert, Pressable, StyleSheet, Text, TextInput, View } from 'react-native';
import { notesRepository } from '@/db/notes-repository';
import { useDebouncedCallback } from '@/hooks/use-debounced-callback';

export default function NoteEditor() {
  const db = useSQLiteContext();
  const { id } = useLocalSearchParams<{ id: string }>();
  const noteId = Number(id);
  const [title, setTitle] = useState('');
  const [body, setBody] = useState('');
  const [status, setStatus] = useState<'cargando' | 'guardado' | 'editando'>('cargando');

  useEffect(() => {
    notesRepository.get(db, noteId).then((note) => {
      if (!note) return router.back(); // id inválido o nota eliminada
      setTitle(note.title);
      setBody(note.body);
      setStatus('guardado');
    });
  }, [db, noteId]);

  const persist = useDebouncedCallback(async (t: string, b: string) => {
    await notesRepository.update(db, noteId, { title: t, body: b });
    setStatus('guardado');
  });

  function change(t: string, b: string) {
    setTitle(t);
    setBody(b);
    setStatus('editando');
    persist(t, b);
  }

  function remove() {
    Alert.alert('Eliminar nota', 'Esta acción no se puede deshacer.', [
      { text: 'Cancelar', style: 'cancel' },
      { text: 'Eliminar', style: 'destructive', onPress: () => notesRepository.remove(db, noteId).then(() => router.back()) },
    ]);
  }

  return (
    <View style={styles.container}>
      <Stack.Screen
        options={{
          headerRight: () => (
            <Pressable onPress={remove} hitSlop={8} accessibilityLabel="Eliminar nota">
              <Text style={styles.delete}>Eliminar</Text>
            </Pressable>
          ),
        }}
      />
      <Text style={styles.status}>{status === 'editando' ? 'Guardando…' : status === 'guardado' ? 'Guardado' : ''}</Text>
      <TextInput value={title} onChangeText={(t) => change(t, body)} placeholder="Título" style={styles.title} />
      <TextInput value={body} onChangeText={(b) => change(title, b)} placeholder="Escribe tu nota…" multiline style={styles.body} textAlignVertical="top" />
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, padding: 20, gap: 8, backgroundColor: '#fff' },
  delete: { color: '#dc2626', fontSize: 17 },
  status: { color: '#94a3b8', fontSize: 12 },
  title: { fontSize: 24, fontWeight: '700' },
  body: { flex: 1, fontSize: 17, lineHeight: 24 },
});
```

## Qué has aprendido

- **Navegación por archivos** con Expo Router: rutas estáticas, dinámicas (`nota/[id]`) y modales.
- **Parámetros de ruta** con `useLocalSearchParams` y su validación (una nota inexistente vuelve atrás).
- **SQLite** con migraciones versionadas y un repositorio que oculta el SQL a las pantallas.
- **`useFocusEffect`** para recargar datos al volver a una pantalla.
- **Debounce** para guardar automáticamente sin escribir en la base de datos en cada tecla.
